==> http.conf <==
# Managed by Runova. Each apply writes a new release, so change this in the app instead.

# WebSocket requests ask the upstream to switch protocols; other requests close as usual.
map $http_upgrade $runova_connection_upgrade {
    default upgrade;
    "" close;
}

# Requests for names no site claims get no answer at all, and TLS handshakes for them
# are refused before any certificate is shown.
server {
    listen 80 default_server;
    listen [::]:80 default_server;
    listen 443 ssl default_server;
    listen [::]:443 ssl default_server;
    server_name _;
    ssl_reject_handshake on;
    return 444;
}

include /etc/nginx/runova/releases/1/sites/cache.conf;

==> sites/cache.conf <==
# Site cache: cache.test

proxy_cache_path /var/cache/nginx/runova-cache levels=1:2 keys_zone=runova_cache_cache:10m max_size=256m inactive=3600s use_temp_path=off;

server {
    listen 80;
    listen [::]:80;
    server_name cache.test;
    server_tokens off;
    root /var/www/runova/sites/cache;
    access_log /var/log/nginx/runova-cache.access.log;
    error_log /var/log/nginx/runova-cache.error.log;

    location ^~ /.well-known/acme-challenge/ {
        root /var/www/runova/acme;
        default_type text/plain;
        try_files $uri =404;
        allow all;
        auth_basic off;
    }

    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        # The visitor's own address replaces any X-Forwarded-For they sent, so it cannot be forged.
        proxy_set_header X-Forwarded-For $remote_addr;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;

        proxy_cache runova_cache_cache;
        proxy_cache_key $scheme$host$request_uri;
        proxy_cache_valid 200 301 302 600s;
        proxy_cache_use_stale error timeout updating http_500 http_502 http_503 http_504;
        proxy_cache_lock on;
        proxy_cache_bypass $http_authorization $cookie_session;
        proxy_no_cache $http_authorization $cookie_session;
        add_header X-Cache-Status $upstream_cache_status always;

        # nginx drops the server's add_header lines in a location that has its own, so they repeat here.
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    }
}
